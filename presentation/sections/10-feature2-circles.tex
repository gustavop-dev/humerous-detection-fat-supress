% Slide 10 -- Feature 2: per-slice articular circles (text only)
\begin{frame}{Feature 2: per-slice articular circles}
  \begin{block}{Key idea}
    The circles, as computed by K\aa{}sa, contain arcs that should \alert{lie on the articular surface},
    providing a \alert{validation method} of the approximating sphere and a useful reference
    for the glenohumeral space.
  \end{block}

  \begin{block}{How they are computed}
    \begin{itemize}
      \item Articular slices: mask area $\geq 70\%$ of the peak.
      \item Project boundary to the slice's best-fit plane via \alert{SVD}.
      \item Iterative 2D RANSAC + K\aa{}sa fit (up to 3 circles per slice).
      \item Inlier distance $\leq 1.5$\,mm; ratio $\geq 35\%$; min.~10 inliers.
    \end{itemize}
  \end{block}
\end{frame}

% Slide 10c -- What each circle stores + clinical meaning (text only)
\begin{frame}{Feature 2: stored information and clinical meaning}
  \begin{block}{What each circle stores}
    \begin{itemize}
      \item Center in 3D and radius in mm units (not pixels).
      \item In-plane orthonormal basis $(\mathbf{u},\mathbf{v})$ and slice normal $\mathbf{n}$.
      \item Inlier point cloud and residual statistics.
      \item Re-projection on the original axial PNG for slice-by-slice clinician review.
    \end{itemize}
  \end{block}

  \begin{block}{Clinical interpretation}
    \begin{itemize}
      \item A focal \alert{drop in inlier ratio} on a circle indicates loss of articular continuity.
      \item A sudden \alert{shift of the center} or radius vs neighbors flags asymmetric pathology.
      \item Each circle is rendered as a \texttt{*\_circle.png} so the radiologist can verify each fit.
    \end{itemize}
  \end{block}
\end{frame}
