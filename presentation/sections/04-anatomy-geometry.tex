% Slide 4 -- Anatomy to geometry: two primitives
\begin{frame}{From anatomy to geometry: two natural primitives}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{The humeral head $\approx$ spherical cap}
        \begin{itemize}
          \item Articular cartilage covers a near-perfect spherical surface (radius $\sim$23--26\,mm).
          \item Motivates a \alert{global sphere fit} as a patient-specific anatomical reference.
        \end{itemize}
      \end{block}
      \begin{block}{The articular surface $\approx$ local circles}
        \begin{itemize}
          \item In equatorial axial slices, the contact contour with the glenoid is locally circular.
          \item Motivates \alert{per-slice circles} that \emph{lie on} the articular surface.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.75]
        \draw[thick, medBlue, fill=medSoft] (0,0) ellipse (1.6 and 1.2);

    \end{column}
  \end{columns}
\end{frame}

% Slide 4b -- Anatomical asymmetry
\begin{frame}{Anatomical asymmetry: head vs shaft}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{The two ends of the humerus look different}
        \begin{itemize}
          \item \textbf{Head}: closes \alert{abruptly} at the anatomical neck.
          \item \textbf{Shaft}: tapers \alert{gradually} (longer, smoother).
        \end{itemize}
      \end{block}
      \begin{block}{Direct consequence on the pipeline}
        Two asymmetric thresholds in the area-curve validation:
        \begin{itemize}
          \item \alert{30\%} of peak on the \textbf{head} side.
          \item \alert{10\%} of peak on the \textbf{shaft} side.
        \end{itemize}
        Anatomy, not arbitrary heuristics.
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.85]
        \draw[->] (0,0) -- (5.5,0) node[right, font=\tiny] {slice};
        \draw[->] (0,0) -- (0,3.5) node[above, font=\tiny] {area};
        \draw[thick, medBlue, smooth, samples=80, domain=0.3:5.0]
          plot (\x, {2.6*exp(-pow(\x-2.5,2)/1.4)});
        \draw[medAccent, dashed] (0.6,0) -- (0.6,1.0)
          node[above, font=\tiny, medAccent] {30\% (head)};
        \draw[medAccent, dashed] (4.6,0) -- (4.6,0.4)
          node[above, font=\tiny, medAccent] {10\% (shaft)};
        \node[medTeal, font=\tiny] at (0.6,2.8) {H $\rightarrow$};
        \node[medTeal, font=\tiny] at (4.6,2.8) {$\leftarrow$ shaft};
      \end{tikzpicture}
      \\[0.3em]
      {\footnotesize Asymmetric trim thresholds.}
    \end{column}
  \end{columns}
\end{frame}
