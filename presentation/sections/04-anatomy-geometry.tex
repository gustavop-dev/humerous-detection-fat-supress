% Slide 4 -- From anatomy to geometry (consolidated)
\begin{frame}{From anatomy to geometry}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{Two anatomical facts $\Rightarrow$ two geometric features}
        \begin{itemize}
          \item Humeral head $\approx$ \alert{sphere} (radius 23--26 mm).
          \item Articular contour $\approx$ \alert{circles} that \emph{lie on} the surface, slice by slice.
        \end{itemize}
      \end{block}
      \begin{block}{Head $\neq$ shaft}
        \begin{itemize}
          \item Head closes \alert{abruptly}; shaft tapers \alert{gradually}.
          \item $\Rightarrow$ asymmetric trim: \alert{30\%} on the head side, \alert{10\%} on the shaft side.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.95]
        \draw[->] (0,0) -- (6.0,0) node[right, font=\scriptsize] {slice};
        \draw[->] (0,0) -- (0,3.5) node[above, font=\scriptsize] {area};
        \draw[thick, medBlue, smooth, samples=80, domain=0.3:5.7]
          plot (\x, {2.8*exp(-pow(\x-3.0,2)/1.4)});
        \draw[medAccent, dashed] (0.7,0) -- (0.7,0.9)
          node[above, font=\scriptsize, medAccent] {30\%};
        \draw[medAccent, dashed] (5.2,0) -- (5.2,0.4)
          node[above, font=\scriptsize, medAccent] {10\%};
        \node[medTeal, font=\scriptsize] at (0.7,3.0) {head};
        \node[medTeal, font=\scriptsize] at (5.2,3.0) {shaft};
      \end{tikzpicture}
      \\[0.3em]
      {\scriptsize Asymmetric trim, driven by anatomy.}
    \end{column}
  \end{columns}
\end{frame}
