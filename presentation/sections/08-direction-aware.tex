% Slide 8 -- Direction-aware validation
\begin{frame}{Stage 2: Direction-aware validation of the area curve}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Reading anatomy from the DICOM header}
        \begin{itemize}
          \item Compute slice normal $\mathbf{n} = \mathrm{row} \times \mathrm{col}$ from \texttt{ImageOrientationPatient}.
          \item Project \texttt{ImagePositionPatient} on $\mathbf{n}$ $\Rightarrow$ \alert{which end is the head}.
          \item Fallback: \texttt{SliceLocation} $\to$ \texttt{InstanceNumber}.
        \end{itemize}
      \end{block}

      \begin{block}{Asymmetric trim (anatomy-driven)}
        \begin{itemize}
          \item Static floor from each border:
            \begin{itemize}\footnotesize
              \item \alert{30\% of peak} on the head side (abrupt closure)
              \item \alert{10\% of peak} on the shaft side (gradual taper)
            \end{itemize}
          \item Reversal-from-peak: stop when the curve genuinely rises ($\geq$10\% head, $\geq$20\% shaft, $\geq$50\,px absolute).
          \item Internal-outlier flagging: robust MAD residual ($\geq 2.5\sigma$) cross-checked with circle inlier-ratio drop ($\geq$15\%) or Dice drop ($\geq$10\%).
        \end{itemize}
      \end{block}
    \end{column}
