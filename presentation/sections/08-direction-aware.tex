% Slide 8 -- Direction-aware validation: header reading (text only)
\begin{frame}{Stage 2: direction-aware validation of the area curve}
  \begin{block}{Reading anatomy from the DICOM header}
    \begin{itemize}
      \item Compute slice normal $\mathbf{n} = \mathrm{row} \times \mathrm{col}$ from \texttt{ImageOrientationPatient}.
      \item Project \texttt{ImagePositionPatient} on $\mathbf{n}$ $\Rightarrow$ \alert{which end is the head}.
      \item Fallback: \texttt{SliceLocation} $\to$ \texttt{InstanceNumber}.
    \end{itemize}
  \end{block}

  \begin{block}{Why the headers, not conventions}
    Vendor conventions about scan direction are inconsistent. Reading the geometric headers makes the pipeline \alert{vendor-agnostic by construction}.
  \end{block}
\end{frame}

% Slide 8c -- Asymmetric trim and outlier flagging (text only)
\begin{frame}{Asymmetric trim and outlier flagging}
  \begin{block}{Asymmetric static floor (anatomy-driven)}
    \begin{itemize}
      \item \alert{30\% of peak} on the head side --- abrupt closure at the anatomical neck.
      \item \alert{10\% of peak} on the shaft side --- gradual taper.
    \end{itemize}
  \end{block}

  \begin{block}{Reversal-from-peak}
    Walk outward from the peak; trim when the curve genuinely rises again ($\geq$10\% on the head side, $\geq$20\% on the shaft side, $\geq$50\,px absolute).
  \end{block}

  \begin{block}{Internal outlier flagging}
    Robust MAD residual ($\geq 2.5\sigma$), cross-checked with circle inlier-ratio drop ($\geq$15\%) or Dice drop ($\geq$10\%).
    Strong outliers excluded; weak ones reported but kept.
  \end{block}
\end{frame}
