% Slide 9 -- Feature 1: sphere (definition + computation)
\begin{frame}{Feature 1: global humeral-head sphere}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{What it is}
        A single sphere fitted to the 3D contour cloud (DICOM millimeters):
        a patient-specific \alert{anatomical reference} for the humeral head.
      \end{block}

      \begin{block}{How it is computed}
        \begin{itemize}
          \item Pixel $\to$ world via \texttt{IPP}, \texttt{IOP}, \texttt{PixelSpacing}.
          \item \alert{RANSAC sphere}: 2000 iterations, 4-point sampling, K\aa{}sa fit.
          \item Inlier distance $\leq 2.0$\,mm; final refit by least-squares on inliers.
          \item Sanity bound: $r < 200$\,mm.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.95]
        \draw[thick, medBlue, fill=medSoft, fill opacity=0.6] (0,0) circle (1.8);

% Slide 9b -- Why it is clinically useful
\begin{frame}{Feature 1: clinical interpretation}
  \begin{block}{Patient-specific anatomical reference}
    \begin{itemize}
      \item Recovers the \alert{expected curvature} of the humeral head (typical $r \in 23$--$26$\,mm in adults).
      \item Centroid $\mathbf{c}$ + radius $r$ characterize the ``ideal'' humeral head for this patient.
      \item Outliers (tubercles, shaft, artifacts) are excluded \alert{automatically} by RANSAC.
    \end{itemize}
  \end{block}

  \begin{block}{Why it matters clinically}
    \begin{itemize}
      \item Enables \alert{normalized comparison} across patients, vendors, and acquisitions.
      \item Provides a per-patient \emph{baseline geometry} against which local deviations stand out.
      \item Sanity-checks downstream features: a per-slice circle far from the global sphere is suspicious.
    \end{itemize}
  \end{block}
\end{frame}
