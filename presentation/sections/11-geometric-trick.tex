% Slide 11 -- The geometric trick: angular sectors -> tendons
\begin{frame}{The geometric trick: angular sectors $\to$ rotator cuff}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Parameterizing the articular circle}
        Each per-slice circle has an in-plane basis $(\mathbf{u},\mathbf{v})$ and a center $\mathbf{c}$.
        Any inlier $\mathbf{p}$ has angular coordinate
        \[
          \theta = \mathrm{atan2}\big((\mathbf{p}-\mathbf{c})\!\cdot\!\mathbf{v},\; (\mathbf{p}-\mathbf{c})\!\cdot\!\mathbf{u}\big).
        \]
      \end{block}

      \begin{block}{Anatomical sector mapping}
        \footnotesize Tendon insertions fall in \alert{predictable angular sectors}:
        \begin{itemize}\footnotesize
          \item \textbf{Subscapularis} $\to$ lesser tub.: $\sim 0$--$45^\circ$.
          \item \textbf{Supraspinatus} $\to$ greater tub. (sup): $\sim 45$--$135^\circ$.
          \item \textbf{Infraspinatus} $\to$ greater tub. (post): $\sim 135$--$225^\circ$.
          \item \textbf{Teres minor} $\to$ greater tub. (inf-post): $\sim 225$--$270^\circ$.
        \end{itemize}
      \end{block}

      \begin{block}{Localization rule}
        A focal drop in inlier ratio in a specific sector $\to$ candidate tear in the corresponding tendon, at the slice height where the drop occurs.
      \end{block}
    \end{column}
