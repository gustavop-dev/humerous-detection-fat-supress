% Slide 11 -- The geometric trick (consolidated to 2 frames)
\begin{frame}{The geometric trick: the circle as a tendon map}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{From a circle to one angle}
        Each per-slice circle has its own \alert{$(\mathbf{u},\mathbf{v})$ axes} on the slice plane.
        Every inlier point on the circle has a single angle $\theta$ around the center:
        \[
          \theta = \mathrm{atan2}\big((\mathbf{p}-\mathbf{c})\!\cdot\!\mathbf{v},\; (\mathbf{p}-\mathbf{c})\!\cdot\!\mathbf{u}\big).
        \]
      \end{block}
      \begin{block}{Why this is powerful}
        The same $\theta$ corresponds to the \alert{same anatomical direction} on the humeral head --- for every patient and every vendor.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \begin{tikzpicture}[scale=1.0]
        \filldraw[medSoft] (0,0) circle (1.8);
        \draw[thick, medBlue] (0,0) circle (1.8);
        \draw[->] (-2.3,0) -- (2.3,0) node[right, font=\scriptsize] {$\mathbf{u}$};
        \draw[->] (0,-2.3) -- (0,2.3) node[above, font=\scriptsize] {$\mathbf{v}$};
        \filldraw[medBlue] (0,0) circle (1.5pt);
        \node[medBlue, font=\scriptsize, below left] at (0,0) {$\mathbf{c}$};
        \filldraw[medAccent] (60:1.8) circle (2pt);
        \draw[medAccent, dashed] (0,0) -- (60:1.8);
        \node[medAccent, font=\scriptsize, above right] at (60:1.8) {$\mathbf{p}$};
        \draw[medAccent] (0.55,0) arc (0:60:0.55);
        \node[medAccent, font=\scriptsize] at (35:0.78) {$\theta$};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

% Slide 11b -- Angular sectors map to tendons
\begin{frame}{Angular sectors $\to$ rotator-cuff tendons}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Anatomical mapping}
        Each tendon inserts in a \alert{predictable angular sector}:
        \begin{itemize}\small
          \item Subscapularis: $\sim 0$--$45^\circ$
          \item Supraspinatus: $\sim 45$--$135^\circ$
          \item Infraspinatus: $\sim 135$--$225^\circ$
          \item Teres minor: $\sim 225$--$270^\circ$
        \end{itemize}
      \end{block}
      \begin{block}{Localization rule}
        A focal drop in inlier ratio inside a sector $\Rightarrow$ a candidate \alert{tear in that tendon}, at the slice where the drop occurs.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \begin{tikzpicture}[scale=1.1]
        \filldraw[medSoft] (0,0) circle (2.0);
        \draw[thick, medBlue] (0,0) circle (2.0);

      \end{tikzpicture}
      \\[0.3em]
      {\scriptsize The articular circle becomes a 1D \emph{tendon map}.}
    \end{column}
  \end{columns}
\end{frame}
