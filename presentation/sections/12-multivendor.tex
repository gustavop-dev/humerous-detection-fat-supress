\begin{frame}{Multi-vendor robustness and equity of access}
  \begin{block}{Tested across vendors and sites}
    \begin{itemize}
      \item \textbf{SIEMENS} --- Quibd\'o, PD-fs TSE.
      \item \textbf{GE} --- Optima MR360, Bogot\'a, PD-fs FSE.
      \item \textbf{Philips} --- Ingenia, SOMA, ePDW SPAIR.
      \item Matrix 256$\times$256 to 320$\times$320; PD-fs and T1-fs.
    \end{itemize}
  \end{block}

  \begin{block}{Why this generalizes}
    \begin{itemize}
      \item Frozen SAM $\Rightarrow$ no domain-specific weights to overfit.
      \item Thresholds are \alert{geometric} (mm, \% of peak), not intensity-based.
      \item Direction handling reads DICOM headers, not vendor conventions.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Equity of access: why this matters in Colombia}
  \begin{block}{The setting}
    Datasets include \alert{Quibd\'o, Choc\'o}, a region with limited subspecialty radiology coverage despite the presence of modern MRI scanners.
  \end{block}

  \begin{block}{The opportunity}
    \begin{itemize}
      \item A vendor-agnostic, interpretable AI tool can directly support diagnostic capacity in such settings.
      \item Geometric outputs are \alert{auditable} by generalist radiologists, lowering the barrier compared to opaque deep-learning classifiers.
      \item Aligns with the UEMS pillar of \textbf{Impacting} medical practice across Europe and beyond.
    \end{itemize}
  \end{block}
\end{frame}
