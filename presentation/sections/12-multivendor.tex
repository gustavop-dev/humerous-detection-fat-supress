\begin{frame}{Multi-vendor robustness and equity of access}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{Tested across 3 vendors and 3 sites}
        \begin{itemize}\small
          \item \textbf{SIEMENS} --- Quibd\'o (Choc\'o), PD-fs / T1-fs.
          \item \textbf{GE} --- Optima MR360, Bogot\'a, PD-fs.
          \item \textbf{Philips} --- Ingenia, SOMA Radiology.
        \end{itemize}
        \emph{6 datasets, matrices 256$\times$256 to 320$\times$320.}
      \end{block}
      \begin{block}{Why it generalizes}
        Frozen SAM (no overfitting), \alert{geometric} thresholds (mm, \% of peak), direction from DICOM headers.
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Why this matters in Colombia}
        Regions like \alert{Choc\'o} have modern MRI but few subspecialty radiologists. An auditable, interpretable tool can directly support generalist readers --- aligned with the UEMS pillar of \textbf{impacting} medical practice.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
