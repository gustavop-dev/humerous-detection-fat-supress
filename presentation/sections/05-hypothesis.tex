% Slide 5 -- Hypothesis and objective (consolidated, design principles inlined)
\begin{frame}{Hypothesis and objective}
  \begin{block}{Hypothesis}
    A tear leaves a \alert{focal, geometric trace} on the articular contour. If we extract clinically
    meaningful geometric features, we can \alert{localize} the lesion to a specific tendon ---
    \emph{without} training a new neural network.
  \end{block}

  \begin{block}{What we built}
    An end-to-end pipeline that is:
    \begin{itemize}
      \item \alert{Interpretable} --- every output is a mask, a circle, or a sphere a radiologist can verify.
      \item \alert{Vendor-agnostic} --- works on SIEMENS, GE and Philips without retraining.
      \item \alert{Anatomy-aware} --- thresholds come from anatomy, not arbitrary heuristics.
    \end{itemize}
  \end{block}
\end{frame}
