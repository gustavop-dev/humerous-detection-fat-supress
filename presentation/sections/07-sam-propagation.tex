\begin{frame}{Stage 1: SAM segmentation + bidirectional propagation}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{Foundation model, no training}
        \begin{itemize}
          \item \alert{SAM (ViT-B)} as a frozen segmenter --- no labels, no fine-tuning.
          \item One seed slice (one click) starts everything.
          \item The mask of slice $i$ provides the prompt for slice $i\!\pm\!1$.
        \end{itemize}
      \end{block}
      \begin{block}{Adaptive quality control}
        Dice threshold is \alert{stricter near the seed, looser far from it}. If a slice fails, a negative prompt is added and the slice is re-segmented.
      \end{block}
    \end{column}
